#define BLYNK_TEMPLATE_ID "YOUR_TEMPLATE_ID"
#define BLYNK_TEMPLATE_NAME "Crop Monitoring System"
#define BLYNK_AUTH_TOKEN "YOUR_AUTH_TOKEN"

#define BLYNK_PRINT Serial

#include <WiFi.h>
#include <BlynkSimpleEsp32.h>
#include <DHT.h>
#include <Wire.h>
#include <Adafruit_BMP280.h>

// ================= WiFi =================

char ssid[] = "YOUR_WIFI_NAME";
char pass[] = "YOUR_WIFI_PASSWORD";

// ================= Sensors =================

// DHT22
#define DHTPIN 4
#define DHTTYPE DHT22

DHT dht(DHTPIN, DHTTYPE);

// BMP280
Adafruit_BMP280 bmp;

// Soil Moisture
#define SOIL_PIN 34

// Rain Sensor
#define RAIN_PIN 35

// ================= Blynk Timer =================

BlynkTimer timer;

// ================= Send Sensor Data =================

void sendSensorData()
{
  // -------- Soil Moisture --------

  int soilRaw = analogRead(SOIL_PIN);

  // Convert to percentage
  int soilMoisture = map(soilRaw, 4095, 0, 0, 100);

  soilMoisture = constrain(soilMoisture, 0, 100);

  // -------- Rain Sensor --------

  int rainRaw = analogRead(RAIN_PIN);

  int rainValue = map(rainRaw, 4095, 0, 0, 100);

  rainValue = constrain(rainValue, 0, 100);

  // -------- DHT22 --------

  float temperature = dht.readTemperature();
  float humidity = dht.readHumidity();

  // -------- BMP280 --------

  float pressure = bmp.readPressure() / 100.0F;

  // -------- Serial Monitor --------

  Serial.println("================================");

  Serial.print("Soil Moisture: ");
  Serial.print(soilMoisture);
  Serial.println("%");

  Serial.print("Temperature: ");
  Serial.print(temperature);
  Serial.println(" °C");

  Serial.print("Humidity: ");
  Serial.print(humidity);
  Serial.println(" %");

  Serial.print("Pressure: ");
  Serial.print(pressure);
  Serial.println(" hPa");

  Serial.print("Rain: ");
  Serial.print(rainValue);
  Serial.println("%");

  // -------- Send to Blynk --------

  Blynk.virtualWrite(V0, soilMoisture);
  Blynk.virtualWrite(V1, temperature);
  Blynk.virtualWrite(V2, humidity);
  Blynk.virtualWrite(V3, pressure);
  Blynk.virtualWrite(V4, rainValue);
}

// ================= Setup =================

void setup()
{
  Serial.begin(115200);

  // Start DHT
  dht.begin();

  // Start I2C
  Wire.begin();

  // Start BMP280
  if (!bmp.begin(0x76))
  {
    Serial.println("BMP280 not found!");

    while (1)
    {
      delay(100);
    }
  }

  // Connect to Blynk
  Blynk.begin(BLYNK_AUTH_TOKEN, ssid, pass);

  // Send data every 2 seconds
  timer.setInterval(2000L, sendSensorData);

  Serial.println("Crop Monitoring System Started!");
}

// ================= Main Loop =================

void loop()
{
  Blynk.run();
  timer.run();
}
